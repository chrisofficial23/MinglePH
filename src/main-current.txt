function setStatus(message, type = '') { const el = $('#setup-status'); el.textContent = message; el.className = `status-line ${type}`; }
function setCallStatus(message, show = true) { $('#call-status').textContent = message; $('#call-status').classList.toggle('hidden', !show); }
function showChat() { $('#chat-panel').classList.remove('hidden'); $('#chat-panel').scrollIntoView({ behavior: 'smooth', block: 'start' }); }
function nickname() { return $('#nickname').value.trim().replace(/[<>]/g, '').slice(0, 24) || `Mingle${Math.floor(1000 + Math.random() * 9000)}`; }
function topicKey() { return (state.topics.length ? state.topics : ['general']).map(t => t.toLowerCase().replace(/[^a-z0-9-]/g, '').slice(0, 24)).filter(Boolean).sort().join('-').slice(0, 90) || 'general'; }
function addTopic() {
  const value = $('#topic-input').value.trim().replace(/^#+/, '').replace(/[<>]/g, '').slice(0, 30);
  if (!value) return;
  if (state.topics.length >= 5) return setStatus('You can add up to 5 topics.', 'error');
  if (state.topics.some(t => t.toLowerCase() === value.toLowerCase())) return setStatus('That topic is already added.', 'error');
  state.topics.push(value); $('#topic-input').value = ''; renderTopics(); setStatus('');
}
function renderTopics() {
  $('#topic-count').textContent = `${state.topics.length}/5 ADDED`;
  const list = $('#topic-list'); list.replaceChildren();
  if (!state.topics.length) { const hint = document.createElement('span'); hint.className = 'topic-hint'; hint.textContent = 'Add up to 5 topics to help find your vibe.'; list.append(hint); return; }
  state.topics.forEach((topic, i) => { const b = document.createElement('button'); b.type = 'button'; b.className = 'topic-chip'; b.textContent = `#${topic} ×`; b.title = 'Remove topic'; b.addEventListener('click', () => { state.topics.splice(i, 1); renderTopics(); }); list.append(b); });
}

async function ensureAuth() {
  if (!configured) throw new Error('Firebase is not configured yet. Open src/main.js and add your Firebase web config. See README.md.');
  if (auth.currentUser) { state.user = auth.currentUser; return state.user; }
  const result = await signInAnonymously(auth); state.user = result.user; return state.user;
}
function clearListeners() { state.listeners.forEach(fn => { try { fn(); } catch {} }); state.listeners = []; }
function addListener(unsub) { state.listeners.push(unsub); }
function resetChatView() {
  $('#chat-messages').innerHTML = '<div class="empty-chat"><div class="loading-heart">♡</div><b>Ready when you are</b><span>Start a chat to meet someone new.</span></div>';
  $('#message-input').disabled = true; $('#send-message').disabled = true;
  $('#chat-title').textContent = 'Chat'; $('#chat-subtitle').textContent = 'Your conversation will appear here.';
  $('#chat-status').innerHTML = '<i></i> Not connected';
}
function appendMessage(data, mine) {
  const node = document.createElement('div'); node.className = `message${mine ? ' mine' : ''}`;
  const author = document.createElement('small'); author.textContent = data.nickname || 'Mingle';
  const body = document.createElement('div'); body.textContent = data.text || '';
  node.append(author, body); $('#chat-messages').append(node); $('#chat-messages').scrollTop = $('#chat-messages').scrollHeight;
}
function roomCollection() { return state.mode === 'group' ? collection(db, 'groupRooms', state.roomId, 'messages') : collection(db, 'chatRooms', state.roomId, 'messages'); }
function watchMessages() {
  const q = query(roomCollection(), limit(100));
  addListener(onSnapshot(q, snapshot => {
    const messages = snapshot.docs.map(d => ({ id: d.id, ...d.data() })).sort((a,b) => (a.createdAt?.toMillis?.() || 0) - (b.createdAt?.toMillis?.() || 0));
    const box = $('#chat-messages'); box.replaceChildren();
    if (!messages.length) { const empty = document.createElement('div'); empty.className = 'empty-chat'; empty.innerHTML = '<div class="loading-heart">♡</div><b>You’re connected!</b><span>Walang hiya-hiya, ikaw na mauna!😉</span>'; box.append(empty); }
    messages.forEach(m => appendMessage(m, m.uid === state.user?.uid));
  }, error => setCallStatus(`Message listener error: ${error.message}`)));
}
function activateChat(title, subtitle, mode) {
  state.mode = mode; showChat(); $('#chat-title').textContent = title; $('#chat-subtitle').textContent = subtitle; $('#chat-status').innerHTML = '<i></i> Connected';
  $('#message-input').disabled = false; $('#send-message').disabled = false;
  const oneToOne = mode === 'one';
  ['#voice-call', '#video-call'].forEach(s => { $(s).disabled = !oneToOne; });
  if (!oneToOne) setCallStatus('Voice and video calls are available in 1-on-1 chats in this version.', true); else setCallStatus('', false);
  watchMessages();
  if (mode === 'one') watchIncomingCalls();
}
async function startGroup() {
  await ensureAuth();
  const key = topicKey(); const ref = doc(db, 'groupRooms', key);
  await setDoc(ref, { type: 'group', topicKey: key, topics: state.topics, active: true, updatedAt: serverTimestamp() }, { merge: true });
  state.roomId = key; state.mode = 'group'; activateChat(`Group: ${state.topics[0] || 'General chika'}`, 'Group room · Anyone with matching topics can join', 'group');
}
async function startOneToOne() {
  const user = await ensureAuth(); state.mode = 'one';
  const myRef = doc(collection(db, 'waitingUsers'));
  const profile = { uid: user.uid, nickname: state.nickname, gender: $('#gender').value, location: $('#location').value, topics: state.topics, status: 'waiting', roomId: null, createdAt: serverTimestamp() };
  await setDoc(myRef, profile); state.queueId = myRef.id;
  showChat(); $('#chat-title').textContent = 'Finding someone...'; $('#chat-subtitle').textContent = 'Keep this page open while we match you.'; $('#chat-status').innerHTML = '<i class="pulse"></i> Searching';
  $('#chat-messages').innerHTML = '<div class="empty-chat"><div class="loading-heart">♡</div><b>Looking for your chat partner...</b><span>Matching may take a moment when few people are online.</span></div>';
  $('#message-input').disabled = true; $('#send-message').disabled = true;

  const waiting = query(collection(db, 'waitingUsers'), where('status', '==', 'waiting'), limit(30));
  const candidates = await getDocs(waiting);
  for (const candidateDoc of candidates.docs) {
    if (candidateDoc.id === myRef.id) continue;
    const candidate = candidateDoc.data();
    if (candidate.uid === user.uid) continue;
    if (candidate.location !== 'Any City / Philippines' && profile.location !== 'Any City / Philippines' && candidate.location !== profile.location) continue;
    const roomRef = doc(collection(db, 'chatRooms')); let matched = false;
    await runTransaction(db, async tx => {
      const mine = await tx.get(myRef); const other = await tx.get(candidateDoc.ref);
      if (!mine.exists() || !other.exists() || mine.data().status !== 'waiting' || other.data().status !== 'waiting') return;
      tx.set(roomRef, { type: 'one', members: [user.uid, candidate.uid], participants: { [user.uid]: { nickname: state.nickname }, [candidate.uid]: { nickname: candidate.nickname || 'Mingle' } }, topics: state.topics, active: true, createdAt: serverTimestamp() });
      tx.update(myRef, { status: 'matched', roomId: roomRef.id }); tx.update(candidateDoc.ref, { status: 'matched', roomId: roomRef.id }); matched = true;
    });
    if (matched) {
      state.roomId = roomRef.id; await deleteDoc(myRef).catch(() => {}); await deleteDoc(candidateDoc.ref).catch(() => {});
      activateChat(`Chat with ${candidate.nickname || 'Mingle'}`, 'You’re connected · Say kumusta!', 'one'); return;
    }
  }
  addListener(onSnapshot(myRef, async snap => {
    if (!snap.exists()) return;
    const data = snap.data();
    if (data.status === 'matched' && data.roomId && !state.roomId) {
      state.roomId = data.roomId; await deleteDoc(myRef).catch(() => {});
      const room = await getDoc(doc(db, 'chatRooms', data.roomId));
      const members = room.exists() ? room.data().members || [] : [];
      const otherUid = members.find(uid => uid !== user.uid); let otherName = 'Mingle';
      if (otherUid && room.exists()) otherName = room.data().participants?.[otherUid]?.nickname || 'Mingle';
      activateChat(`Chat with ${otherName}`, 'You’re connected · Say kumusta!', 'one');
    }
  }));
}

$('#start-chat').addEventListener('click', async () => {
  if (!$('#respect').checked) return setStatus('Please agree to follow the community rules before starting.', 'error');
  if (state.starting) return;
  state.starting = true; $('#start-chat').disabled = true; $('#start-chat').style.opacity = '.65';
  state.nickname = nickname(); setStatus('Connecting…');
  try { clearListeners(); await cleanupCall(false); state.roomId = null; state.queueId = null; if (state.mode === 'group') await startGroup(); else await startOneToOne(); setStatus(''); }
  catch (e) { setStatus(e.message || 'Could not connect. Check your Firebase setup and try again.', 'error'); }
  finally { state.starting = false; $('#start-chat').disabled = false; $('#start-chat').style.opacity = ''; }
});
$('#message-form').addEventListener('submit', async e => {
  e.preventDefault(); const text = $('#message-input').value.trim(); if (!text || !state.roomId || !state.user) return;
  if (text.length > 1000) return;
  $('#message-input').value = '';
  try { await addDoc(roomCollection(), { uid: state.user.uid, nickname: state.nickname, text, createdAt: serverTimestamp() }); }
  catch (err) { setCallStatus(`Message failed: ${err.message}`, true); }
});

// WebRTC voice/video calling. Signaling is stored in Firestore; media remains peer-to-peer.
function setCallButtons(active) {
  $('#voice-call').disabled = !active || state.mode !== 'one'; $('#video-call').disabled = !active || state.mode !== 'one';
  $('#toggle-mic').disabled = !state.localStream; $('#toggle-camera').disabled = !state.localStream; $('#end-call').disabled = !state.pc;
}
async function getMedia(video) {
  if (!navigator.mediaDevices?.getUserMedia) throw new Error('Camera/microphone needs a secure context (HTTPS or localhost) and a supported browser.');
  return navigator.mediaDevices.getUserMedia({ audio: true, video: video ? { facingMode: 'user' } : false });
}
function createPeerConnection() {
  const pc = new RTCPeerConnection({ iceServers: [{ urls: 'stun:stun.l.google.com:19302' }, { urls: 'stun:stun1.l.google.com:19302' }] });
  state.pc = pc; state.remoteStream = new MediaStream(); $('#remote-video').srcObject = state.remoteStream;
  pc.ontrack = e => { e.streams[0]?.getTracks().forEach(track => state.remoteStream.addTrack(track)); $('#video-stage').classList.remove('hidden'); };
  pc.onicecandidate = async e => { if (e.candidate && state.roomId && state.callId) { try { await addDoc(collection(db, 'chatRooms', state.roomId, 'calls', state.callId, 'candidates'), { from: state.user.uid, candidate: e.candidate.toJSON(), createdAt: serverTimestamp() }); } catch (err) { console.warn('Could not send ICE candidate', err); } } };
  pc.onconnectionstatechange = () => { if (['connected', 'connecting'].includes(pc.connectionState)) setCallStatus(pc.connectionState === 'connected' ? 'Call connected. Keep being respectful.' : 'Connecting call…'); if (['failed', 'disconnected'].includes(pc.connectionState)) setCallStatus('Call connection interrupted. Try ending and starting the call again.'); };
  if (state.localStream) state.localStream.getTracks().forEach(track => pc.addTrack(track, state.localStream));
  return pc;
}
async function listenForCandidates() {
  if (state.candidateUnsub) state.candidateUnsub();
  const ref = collection(db, 'chatRooms', state.roomId, 'calls', state.callId, 'candidates');
  state.candidateUnsub = onSnapshot(ref, snapshot => snapshot.docChanges().forEach(change => {
    if (change.type !== 'added') return; const data = change.doc.data();
    if (data.from === state.user.uid || !data.candidate || !state.pc) return;
    state.pc.addIceCandidate(new RTCIceCandidate(data.candidate)).catch(err => console.warn('ICE candidate issue', err));
  }));
}
async function beginCall(video) {
  if (state.mode !== 'one' || !state.roomId) return setCallStatus('Start a 1-on-1 chat before calling.');
  if (state.pc) await cleanupCall(true);
  try {
    state.localStream = await getMedia(video); state.cameraEnabled = video; state.micEnabled = true;
    $('#local-video').srcObject = state.localStream; $('#video-stage').classList.remove('hidden');
    $('#toggle-mic').textContent = 'Mute mic'; $('#toggle-camera').textContent = video ? 'Camera off' : 'Camera on';
    const pc = createPeerConnection(); state.callId = `${state.user.uid}_${Date.now()}`; state.callRole = 'caller';
    const callRef = doc(db, 'chatRooms', state.roomId, 'calls', state.callId);
    const offer = await pc.createOffer(); await pc.setLocalDescription(offer);
    await setDoc(callRef, { caller: state.user.uid, mode: video ? 'video' : 'voice', offer: { type: offer.type, sdp: offer.sdp }, status: 'ringing', createdAt: serverTimestamp() });
    setCallButtons(true); $('#end-call').disabled = false; setCallStatus(video ? 'Video call started. Waiting for your partner to accept/connect…' : 'Voice call started. Waiting for your partner to connect…');
    await listenForCandidates();
    state.callUnsub = onSnapshot(callRef, async snap => {
      if (!snap.exists()) return; const data = snap.data();
      if (data.status === 'ended') { await cleanupCall(false); setCallStatus('The call has ended.'); return; }
      if (data.answer && state.pc?.signalingState === 'have-local-offer') {
        try { await state.pc.setRemoteDescription(new RTCSessionDescription(data.answer)); setCallStatus('Call answer received. Connecting…'); }
        catch (err) { console.warn(err); }
      }
    });
  } catch (err) { await cleanupCall(false); setCallStatus(err.message || 'Could not start the call. Please allow camera/microphone access.'); }
}
async function acceptIncomingCall(callId, data) {
  if (state.handlingOffer || state.pc || data.caller === state.user.uid || !data.offer || data.status === 'ended') return;
  state.handlingOffer = true;
  try {
    const video = data.mode === 'video'; state.localStream = await getMedia(video); state.cameraEnabled = video; state.micEnabled = true;
    $('#local-video').srcObject = state.localStream; $('#video-stage').classList.remove('hidden');
    const pc = createPeerConnection(); state.callId = callId; state.callRole = 'callee';
    await pc.setRemoteDescription(new RTCSessionDescription(data.offer));
    const answer = await pc.createAnswer(); await pc.setLocalDescription(answer);
    const callRef = doc(db, 'chatRooms', state.roomId, 'calls', callId);
    await updateDoc(callRef, { answer: { type: answer.type, sdp: answer.sdp }, status: 'answered', answerer: state.user.uid });
    await listenForCandidates(); setCallButtons(true); $('#end-call').disabled = false;
    $('#toggle-mic').textContent = 'Mute mic'; $('#toggle-camera').textContent = video ? 'Camera off' : 'Camera on';
    setCallStatus(video ? 'Video call connected.' : 'Voice call connected.');
    state.callUnsub = onSnapshot(callRef, async snap => { if (snap.exists() && snap.data().status === 'ended') { await cleanupCall(false); setCallStatus('The call has ended.'); } });
  } catch (err) { await cleanupCall(false); setCallStatus(`Could not join call: ${err.message}`); }
  finally { state.handlingOffer = false; }
}
function watchIncomingCalls() {
  const callsRef = collection(db, 'chatRooms', state.roomId, 'calls');
  addListener(onSnapshot(callsRef, snap => {
    snap.docChanges().forEach(change => {
      if (change.type !== 'added' && change.type !== 'modified') return;
      const data = change.doc.data();
      if (data.caller !== state.user.uid && data.offer && !data.answer && data.status === 'ringing' && !state.pc) {
        const accept = window.confirm(`${data.mode === 'video' ? 'Video' : 'Voice'} call incoming. Accept?`);
        if (accept) acceptIncomingCall(change.doc.id, data);
        else updateDoc(change.doc.ref, { status: 'declined' }).catch(() => {});
      }
    });
  }));
}
async function cleanupCall(updateRemote = false) {
  if (updateRemote && state.roomId && state.callId && db) {
    try { await updateDoc(doc(db, 'chatRooms', state.roomId, 'calls', state.callId), { status: 'ended', endedAt: serverTimestamp() }); } catch {}
  }
  if (state.callUnsub) state.callUnsub(); state.callUnsub = null;
  if (state.candidateUnsub) state.candidateUnsub(); state.candidateUnsub = null;
  if (state.pc) { state.pc.ontrack = null; state.pc.onicecandidate = null; state.pc.close(); } state.pc = null;
  if (state.localStream) state.localStream.getTracks().forEach(t => t.stop()); state.localStream = null;
  if (state.remoteStream) state.remoteStream.getTracks().forEach(t => t.stop()); state.remoteStream = null;
  $('#local-video').srcObject = null; $('#remote-video').srcObject = null; $('#video-stage').classList.add('hidden');
  state.callId = null; state.callRole = null; state.cameraEnabled = false; state.micEnabled = false;
  $('#toggle-mic').textContent = 'Mic on'; $('#toggle-camera').textContent = 'Camera off';
  setCallButtons(Boolean(state.roomId && state.mode === 'one'));
}

$('#toggle-camera').addEventListener('click', () => { if (!state.localStream) return; state.cameraEnabled = !state.cameraEnabled; state.localStream.getVideoTracks().forEach(t => t.enabled = state.cameraEnabled); $('#toggle-camera').textContent = state.cameraEnabled ? 'Camera off' : 'Camera on'; });

async function leaveChat(requeue = false) {
  await cleanupCall(true); clearListeners();
  if (state.queueId && db) await deleteDoc(doc(db, 'waitingUsers', state.queueId)).catch(() => {});
  if (state.roomId && state.mode === 'one' && state.user && db) {
    await updateDoc(doc(db, 'chatRooms', state.roomId), { active: false, endedAt: serverTimestamp() }).catch(() => {});
  }
  state.roomId = null; state.queueId = null; resetChatView(); setCallStatus('', false);
  if (requeue) $('#start-chat').click();
}

  catch (e) { setCallStatus(`Could not submit report: ${e.message}`); }
});
